Config.Debug = false -- Enable console debug prints
Config.Locale = 'en' -- UI language (15 languages available)
Config.DefaultSpawn = vector4(-1035.71, -2731.87, 12.86, 0.0)

Config.SkipSelection = false -- Auto-spawn/skips external sapwn selectors
Config.AutoCloseLoadingScreen = true --Auto closes Loading screen
Config.EnableRandomScenes = false -- Randomize civilian scenes
